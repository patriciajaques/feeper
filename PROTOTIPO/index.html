package PSP0;

import java.util.ArrayList;
import java.util.Scanner;
import java.lang.Math;

public class programa1 {

	public static void main(String[] args) {

		
		ArrayList<Integer> lista = new ArrayList<Integer>();
		
		//Carregar os dados
		lista.add(186);
		lista.add(699);
		lista.add(132);
		lista.add(272);
		lista.add(291);
		lista.add(331);
		lista.add(199);
		lista.add(1890);
		lista.add(788);
		lista.add(1601);

		
		double desvio = CalculaDesvio(lista);
		
		System.out.println("");
		System.out.println("");
		System.out.println("DESVIO PADRAO: ");
		System.out.println(desvio);
		
		

	}
	
	public static double CalculaMedia(ArrayList<Integer> lista)
	{
		int n = lista.size()
		int soma = 0;
		
		for(int i = 0; i< n; i++)
		{
			soma += lista.get(i);
		}
		return soma / n;
	}
	
	public static double CalculaDesvio(ArrayList<Integer> lista)
	{
		double media = CalculaMedia(lista);
		double desvioLinhas = 0;
		int n = lista.size();
		
		for(int i = 0; i< n; i++)
		{
			double valorDaLinha = (lista.get(i) - media);
			desvioLinhas += valorDaLinha * valorDaLinha;
		}
		
		double divisao = desvioLinhas / (n - 1);
		double raiz = Math.sqrt(divisao);
		return raiz;
	}

}
